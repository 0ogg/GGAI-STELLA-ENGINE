SillyTavern의 `world-info.js` 코드와 저장/로드 로직을 분석하여, 프론트엔드에서 SillyTavern와 호환되는 월드인포(Lorebook) `.json` 파일을 생성하거나 임포트할 수 있도록 핵심 데이터 구조(Key-Value 스키마)를 추출해 드립니다.



SillyTavern에서 월드인포를 내보낼 때 생성되는 JSON 파일은 크게 \*\*루트 메타데이터\*\*와 \*\*엔트리(entries) 딕셔너리\*\*로 구성됩니다.



\---



\### 1. 월드인포 JSON 기본 구조 (Root Structure)



```json

{

&#x20; "name": "월드인포\_이름",

&#x20; "entries": {

&#x20;   "0": { ... },

&#x20;   "1": { ... }

&#x20; }

}

```

\*   `name`: 월드인포(책)의 이름입니다.

\*   `entries`: 각 정보 조각(엔트리)을 담는 객체입니다. \*\*주의할 점은 배열(`\[]`)이 아니라 딕셔너리(`{}`) 형태\*\*이며, 키(Key)는 엔트리의 고유 번호(`uid`)를 문자열로 사용합니다.



\---



\### 2. 엔트리 상세 스키마 (Entry Key-Value)



`entries` 안에 들어가는 개별 엔트리의 필드와 기본값입니다. 프론트엔드에서 객체를 생성할 때 이 속성들을 포함시키면 됩니다.



```javascript

const entryTemplate = {

&#x20;   // === 식별자 ===

&#x20;   "uid": 0,                   // 고유 ID (entries의 키와 동일한 숫자)



&#x20;   // === 키워드 설정 ===

&#x20;   "key": \["키워드1", "키워드2"],  // 메인 키워드 배열 (콤마로 구분하여 입력)

&#x20;   "keysecondary": \[],           // 보조 키워드 배열 (선택적 작동 조건)

&#x20;   "comment": "엔트리 메모/이름",  // UI에 표시되는 엔트리 이름



&#x20;   // === 내용 ===

&#x20;   "content": "여기에 AI에게 주입할 월드빌딩 텍스트를 작성합니다.",



&#x20;   // === 작동 조건 ===

&#x20;   "constant": false,            // 상시 활성화 여부 (키워드 무시하고 항상 삽입)

&#x20;   "selective": false,           // 보조 키워드 논리 적용 여부

&#x20;   "selectiveLogic": 0,          // 0: AND (둘 다), 1: NOT (보조 키워드 제외)

&#x20;   "addMemo": true,              // 엔트리 이름(comment)을 내용 앞에 메모로 추가할지 여부

&#x20;   "order": 100,                 // 우선순위 (숫자가 클수록 먼저 삽입됨)

&#x20;   "position": 1,                // 삽입 위치 (자세한 값은 아래 참조)

&#x20;   "disable": false,             // 비활성화 여부



&#x20;   // === 고급 설정 ===

&#x20;   "excludeRecursion": false,    // 재귀적 트리거(이 엔트리가 다른 엔트리를 트리거하는 것) 방지

&#x20;   "preventRecursion": false,    // 다른 엔트리가 이 엔트리를 트리거하는 것 방지

&#x20;   "delayUntilRecursion": false, // 재귀 트리거 시에만 활성화

&#x20;   "probability": 100,           // 확률적 활성화 (0\~100)

&#x20;   "useProbability": true,       // 확률 적용 여부

&#x20;   "depth": 4,                   // position이 4(at\_depth)일 때의 삽입 깊이

&#x20;   "group": "",                  // 그룹화 식별자

&#x20;   "groupOverride": false,       // 그룹 우선순위 덮어쓰기

&#x20;   "groupWeight": 100,           // 그룹 내 가중치

&#x20;   "scanDepth": null,            // 키워드 스캔 깊이 오버라이드 (null=전역 설정 따름)

&#x20;   "caseSensitive": null,        // 대소문자 구분 오버라이드 (null=전역 설정 따름)

&#x20;   "matchWholeWords": null,      // 전체 단어 일치 오버라이드 (null=전역 설정 따름)

&#x20;   "useGroupScoring": false,     // 그룹 점수 기반 선택 사용

&#x20;   "automationId": "",           // 자동화 규칙 ID



&#x20;   // === 최신 기능 (벡터/역할) ===

&#x20;   "vectorized": false,          // 벡터 임베딩 사용 여부 (로컬 벡터 스토리지)

&#x20;   "role": 0                     // 메시지 역할 (0: system, 1: user, 2: assistant)

};

```



\#### 💡 `position` 값 매핑 (어디에 삽입할 것인가)

\*   `0`: 캐릭터 정의 앞 (Before Char Def)

\*   `1`: 캐릭터 정의 뒤 (After Char Def) - \*\*기본값\*\*

\*   `2`: 예시 메시지 앞 (Before Example Messages)

\*   `3`: 예시 메시지 뒤 (After Example Messages)

\*   `4`: 지정된 깊이에 (At Depth) - `depth` 속성과 함께 사용



\---



\### 3. 프론트엔드에서 JSON 생성 및 다운로드 예제



프론트엔드에서 입력 폼 데이터를 이용해 SillyTavern 호환 JSON을 만들고 다운로드하는 함수입니다.



```javascript

function generateAndDownloadWorldInfo(fileName, entriesData) {

&#x20;   // 1. 뼈대 생성

&#x20;   const worldInfoJson = {

&#x20;       name: fileName,

&#x20;       entries: {}

&#x20;   };



&#x20;   // 2. 엔트리 데이터 매핑 (entriesData는 프론트엔드 폼에서 모은 배열이라 가정)

&#x20;   entriesData.forEach((entry, index) => {

&#x20;       const uid = index; // 고유 ID 생성 (보통 인덱스 사용)

&#x20;       

&#x20;       // 필수 속성이 없을 경우 기본값 채움 (에러 방지)

&#x20;       worldInfoJson.entries\[String(uid)] = {

&#x20;           uid: uid,

&#x20;           key: entry.keys || \[],

&#x20;           keysecondary: entry.secondaryKeys || \[],

&#x20;           comment: entry.name || "",

&#x20;           content: entry.content || "",

&#x20;           constant: entry.constant || false,

&#x20;           selective: entry.selective || false,

&#x20;           selectiveLogic: entry.selectiveLogic || 0,

&#x20;           addMemo: entry.addMemo !== undefined ? entry.addMemo : true,

&#x20;           order: entry.order || 100,

&#x20;           position: entry.position !== undefined ? entry.position : 1,

&#x20;           disable: entry.disable || false,

&#x20;           excludeRecursion: false,

&#x20;           preventRecursion: false,

&#x20;           delayUntilRecursion: false,

&#x20;           probability: 100,

&#x20;           useProbability: true,

&#x20;           depth: entry.depth || 4,

&#x20;           group: "",

&#x20;           groupOverride: false,

&#x20;           groupWeight: 100,

&#x20;           scanDepth: null,

&#x20;           caseSensitive: null,

&#x20;           matchWholeWords: null,

&#x20;           useGroupScoring: false,

&#x20;           automationId: "",

&#x20;           vectorized: false,

&#x20;           role: 0

&#x20;       };

&#x20;   });



&#x20;   // 3. JSON 문자열로 변환

&#x20;   const jsonString = JSON.stringify(worldInfoJson, null, 2);



&#x20;   // 4. 다운로드 트리거

&#x20;   const blob = new Blob(\[jsonString], { type: "application/json" });

&#x20;   const url = URL.createObjectURL(blob);

&#x20;   const a = document.createElement("a");

&#x20;   a.href = url;

&#x20;   a.download = `${fileName}.json`;

&#x20;   document.body.appendChild(a);

&#x20;   a.click();

&#x20;   document.body.removeChild(a);

&#x20;   URL.revokeObjectURL(url);

}



// 사용 예시

const myEntries = \[

&#x20;   { keys: \["마법", "마나"], content: "이 세계의 마법은 마나를 소모한다.", name: "마법 규칙", position: 1, order: 100 },

&#x20;   { keys: \["왕국"], content: "왕국은 300년 전 건국되었다.", constant: true, name: "왕국 역사", position: 0 }

];



generateAndDownloadWorldInfo("나의\_세계관", myEntries);

```



\### 4. 프론트엔드에서 임포트(Import) 하는 방법



사용자가 SillyTavern에서 추출한 `.json`을 프론트엔드로 불러올 때는 `entries`가 객체 형태라는 점을 유의해야 합니다.



```javascript

function handleWorldInfoImport(event) {

&#x20;   const file = event.target.files\[0];

&#x20;   if (!file) return;



&#x20;   const reader = new FileReader();

&#x20;   reader.onload = (e) => {

&#x20;       try {

&#x20;           const data = JSON.parse(e.target.result);

&#x20;           

&#x20;           const worldName = data.name || file.name.replace('.json', '');

&#x20;           const entries = data.entries;



&#x20;           // entries를 배열로 변환하여 프론트엔드 상태(State)에 저장

&#x20;           const entryArray = Object.values(entries).map(entry => ({

&#x20;               id: entry.uid,

&#x20;               keys: entry.key,

&#x20;               secondaryKeys: entry.keysecondary,

&#x20;               name: entry.comment,

&#x20;               content: entry.content,

&#x20;               constant: entry.constant,

&#x20;               position: entry.position,

&#x20;               // ... 필요한 속성만 추출하여 매핑

&#x20;           }));



&#x20;           console.log("임포트된 월드 이름:", worldName);

&#x20;           console.log("임포트된 엔트리 수:", entryArray.length);

&#x20;           // setState(entryArray) 등으로 UI 업데이트



&#x20;       } catch (error) {

&#x20;           console.error("JSON 파싱 에러:", error);

&#x20;           alert("유효하지 않은 월드인포 파일입니다.");

&#x20;       }

&#x20;   };

&#x20;   reader.readAsText(file);

}

```



이 스키마를 사용하면 SillyTavern와 100% 호환되는 월드인포 JSON을 프론트엔드에서 자유롭게 생성하고 읽어올 수 있습니다.

\### 🧱 2. 핵심 데이터 구조 (TypeScript Interface)

AI에게 NovelAI 호환 모듈을 작성시킬 때 이 구조를 제공하세요.



```typescript

interface NovelAILorebook {

&#x20; lorebookVersion: number; // \[필수] 버전을 명시 (보통 5 또는 6)

&#x20; entries: NovelAILorebookEntry\[]; // \[필수] 로어북 항목 배열

&#x20; categories?: NovelAILorebookCategory\[]; // 카테고리 분류 (없을 경우 빈 배열 \[] 처리 권장)

&#x20; settings?: Record<string, any>; // 전역 설정 보존용 (Export 시 원본 유지 목적)

}



interface NovelAILorebookEntry {

&#x20; id: string; // 고유 식별자 (보통 UUID v4 형식)

&#x20; text: string; // 프롬프트에 삽입될 본문 (CCv3의 content)

&#x20; keys: string\[]; // 활성화 키워드 배열 (정규식 또는 일반 텍스트)

&#x20; searchRange: number; // 컨텍스트 스캔 범위 (토큰/문자 수, 보통 1000)

&#x20; enabled: boolean; // 항목 활성화 여부

&#x20; forceActivation: boolean; // true면 조건 상관없이 항상 삽입 (Always On)

&#x20; keyRelative: boolean; 

&#x20; nonStoryActivatable: boolean; 

&#x20; category: string; // 소속된 카테고리의 id (소속이 없다면 빈 문자열 "")

&#x20; 

&#x20; // 삽입 위치 및 토큰 제한(Trim) 제어

&#x20; contextConfig: {

&#x20;   prefix: string; // 기본값 ""

&#x20;   suffix: string; // 기본값 "\\n"

&#x20;   tokenBudget: number; // 사용할 컨텍스트 비율 (ex: 1 = 100%, 0.1 = 10%)

&#x20;   reservedTokens: number; // 항목에 예약할 토큰 수

&#x20;   budgetPriority: number; // 우선순위 (이 숫자가 낮을수록 토큰 제한 시 먼저 버려짐)

&#x20;   trimDirection: 'trimBottom' | 'trimTop' | 'doNotTrim'; // 잘리는 방향

&#x20;   insertionType: 'newline' | 'sentence' | 'token'; 

&#x20;   maximumTrimType: 'sentence' | 'newline' | 'token'; 

&#x20;   insertionPosition: number; // 프롬프트 내 삽입 위치 (음수일 경우 최신 로그 쪽을 기준으로 위치함)

&#x20; };

&#x20; loreBiasGroups?: any\[]; // NovelAI 특유의 단어 확률 보정 (Export 호환을 위해 보존 필요)

}



interface NovelAILorebookCategory {

&#x20; id: string; // 카테고리 고유 식별자 (UUID v4)

&#x20; name: string; // 카테고리 이름

&#x20; enabled: boolean; // 활성화 여부

&#x20; createSubcontext: boolean; // 서브컨텍스트 생성 여부

&#x20; useCategoryConfig?: boolean;

}

```

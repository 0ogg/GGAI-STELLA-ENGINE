interface SillyTavernPromptsExport {

&#x20; version: number; // \[필수] Export 구조 버전 (현재 1)

&#x20; type: 'full' | 'character'; // \[필수] 전체 프롬프트인지 특정 캐릭터용인지 명시

&#x20; data: {

&#x20;   prompts: SillyTavernPrompt\[]; // 프롬프트 객체 배열

&#x20;   prompt\_order: SillyTavernPromptOrderEntry\[]; // 프롬프트 순서 및 활성화 상태 (여기에 있는 순서대로 정렬됨)

&#x20; };

}



interface SillyTavernPrompt {

&#x20; identifier: string; // \[필수] 고유 식별자 (보통 UUID v4. 단, 핵심 프롬프트는 'main', 'jailbreak' 등 고정 문자열 사용)

&#x20; name: string; // 사용자 UI에 표시될 이름

&#x20; role: 'system' | 'user' | 'assistant'; // 전송될 역할

&#x20; content: string; // 프롬프트 본문 내용 ({{char}}, {{user}} 매크로 치환됨)

&#x20; 

&#x20; system\_prompt?: boolean; // 삭제 불가 기본/시스템 프롬프트 여부

&#x20; marker?: boolean; // \[중요] true면 텍스트 삽입용이 아닌 다른 프롬프트를 배치하기 위한 위치 기준점(Anchor) 역할

&#x20; 

&#x20; // 삽입 위치 제어 (INJECTION\_POSITION)

&#x20; // 0: RELATIVE (prompt\_order 순서에 따름) / 1: ABSOLUTE (depth, order 값에 따름)

&#x20; injection\_position?: 0 | 1; 

&#x20; injection\_depth?: number; // 절대 위치일 때 챗 로그 깊이 (기본값: 4)

&#x20; injection\_order?: number; // 절대 위치일 때 삽입 우선순위 (기본값: 100)

&#x20; 

&#x20; forbid\_overrides?: boolean; // 캐릭터 카드에 의해 덮어쓰기 되는 것을 금지할지 여부

&#x20; extension?: boolean; // 확장 기능에 의해 추가된 프롬프트인지 여부

&#x20; injection\_trigger?: string\[]; // 생성 타입 트리거 조건 (예:\['normal', 'continue'])

}



interface SillyTavernPromptOrderEntry {

&#x20; identifier: string; // prompts 배열 내의 특정 프롬프트 identifier 참조

&#x20; enabled: boolean; // 해당 프롬프트의 활성화(체크) 여부

}
